<?php

namespace App\Http\Controllers\Api\Admin;

use App\Enums\UserRole;
use App\Enums\UserStatus;
use App\Http\Controllers\Controller;
use App\Http\Resources\UserResource;
use App\Models\User;
use App\Services\AuditLogger;
use App\Support\ApiResponse;
use App\Support\LikeSearch;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\Rule;
use Illuminate\Validation\ValidationException;

/**
 * Super Admin: user accounts, status and roles (T2.5c).
 * Student accounts cannot change role (they own academic records); roles that need a department
 * or a lecturer profile are refused with a clear message instead of leaving a half-configured user.
 */
class UserController extends Controller
{
    private const WITH = ['department', 'student.department', 'lecturer.department'];

    public function __construct(private readonly AuditLogger $audit)
    {
    }

    /** GET /api/admin/users?search=&role=&status=&page=&per_page= */
    public function index(Request $request): array
    {
        $query = User::query()->with(self::WITH)->orderBy('name');

        LikeSearch::apply($query, ApiResponse::search($request), ['name', 'email']);

        $role = $request->query('role');
        if (is_string($role) && UserRole::tryFrom($role)) {
            $query->where('role', $role);
        }
        $status = $request->query('status');
        if (is_string($status) && UserStatus::tryFrom($status)) {
            $query->where('status', $status);
        }

        return ApiResponse::paginated($query->paginate(ApiResponse::perPage($request)), UserResource::class);
    }

    /** PATCH /api/admin/users/{user}/status  body: {status} */
    public function updateStatus(Request $request, User $user): UserResource
    {
        $data = $request->validate(['status' => ['required', Rule::in(UserStatus::values())]]);

        if ($user->is($request->user())) {
            throw ValidationException::withMessages(['status' => 'You cannot change your own status.']);
        }

        $status = UserStatus::from($data['status']);

        if ($user->status !== $status) {
            DB::transaction(function () use ($request, $user, $status) {
                $user->update(['status' => $status]);

                // A deactivated or suspended user is signed out everywhere straight away.
                if ($status !== UserStatus::Active) {
                    $user->tokens()->delete();
                }

                $this->audit->record($request->user(), 'USER_STATUS_CHANGED', $user->name, "Status set to {$status->value}", $user);
            });
        }

        return new UserResource($user->fresh(self::WITH));
    }

    /** PATCH /api/admin/users/{user}/role  body: {role, departmentId?} */
    public function updateRole(Request $request, User $user): UserResource
    {
        $data = $request->validate([
            'role' => ['required', Rule::in(UserRole::values())],
            'departmentId' => ['nullable', 'integer', 'exists:departments,id'],
        ]);

        if ($user->is($request->user())) {
            throw ValidationException::withMessages(['role' => 'You cannot change your own role.']);
        }

        $from = $user->role;
        $to = UserRole::from($data['role']);

        if ($from === $to) {
            return new UserResource($user->load(self::WITH));
        }

        if ($from === UserRole::Student || $to === UserRole::Student) {
            throw ValidationException::withMessages([
                'role' => 'Student accounts cannot be converted: student records are created and managed from the department pages.',
            ]);
        }

        $user->loadMissing('lecturer');

        if ($to === UserRole::Lecturer && ! $user->lecturer) {
            throw ValidationException::withMessages(['role' => 'This user has no lecturer profile, so they cannot become a lecturer.']);
        }

        // Only department admins are scoped through users.department_id.
        $departmentId = null;
        if ($to === UserRole::DeptAdmin) {
            $departmentId = isset($data['departmentId'])
                ? (int) $data['departmentId']
                : $user->lecturer?->department_id;

            if (! $departmentId) {
                throw ValidationException::withMessages(['departmentId' => 'Choose a department for the new department admin.']);
            }
        }

        DB::transaction(function () use ($request, $user, $to, $departmentId) {
            $user->update(['role' => $to, 'department_id' => $departmentId]);

            $this->audit->record($request->user(), 'USER_ROLE_CHANGED', $user->name, "Role set to {$to->value}", $user);
        });

        return new UserResource($user->fresh(self::WITH));
    }
}

namespace App\Http\Controllers\Api\Admin;

use App\Http\Controllers\Api\Admin\Concerns\ManagesRecords;
use App\Http\Controllers\Controller;
use App\Http\Requests\Admin\ProgramRequest;
use App\Http\Resources\ProgramResource;
use App\Models\Program;
use App\Services\AuditLogger;
use App\Support\ApiResponse;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Http\Response;
use Illuminate\Validation\ValidationException;

/** Super admin CRUD for programs. Optional filter: ?departmentId=. */
class ProgramController extends Controller
{
    use ManagesRecords;

    public function __construct(private AuditLogger $audit) {}

    /** GET /api/admin/programs */
    public function index(Request $request): array
    {
        $query = $this->applySearch(Program::query(), ApiResponse::search($request), ['name', 'award'])
            ->with('department')
            ->withCount('students')
            ->when(ctype_digit((string) $request->query('departmentId')), fn ($q) => $q->where('department_id', (int) $request->query('departmentId')))
            ->orderBy('name');

        return ApiResponse::paginated($query->paginate(ApiResponse::perPage($request)), ProgramResource::class);
    }

    /** GET /api/admin/programs/{program} */
    public function show(Program $program): ProgramResource
    {
        return new ProgramResource($program->load('department')->loadCount('students'));
    }

    /** POST /api/admin/programs */
    public function store(ProgramRequest $request): JsonResponse
    {
        $program = Program::create($request->columns())->load('department');

        $this->audit->record($request->user(), 'RECORD_CREATED', "Program: {$program->name}",
            "Created program {$program->name} ({$program->award}) in {$program->department->name}", $program);

        return (new ProgramResource($program->loadCount('students')))->response()->setStatusCode(201);
    }

    /** PUT /api/admin/programs/{program} */
    public function update(ProgramRequest $request, Program $program): ProgramResource
    {
        $data = $request->columns();

        // Students and courses are tied to the program's department, so it cannot move while they exist.
        if ($data['department_id'] !== (int) $program->department_id
            && ($program->students()->exists() || $program->courses()->exists())) {
            throw ValidationException::withMessages([
                'departmentId' => 'This program has students or courses, so it cannot be moved to another department.',
            ]);
        }

        $changes = $this->applyUpdate($program, $data);
        $program->load('department');

        if ($changes !== []) {
            $this->audit->record($request->user(), 'RECORD_UPDATED', "Program: {$program->name}",
                'Updated program: '.implode(', ', array_keys($changes)), $program, ['changes' => $changes]);
        }

        return new ProgramResource($program->loadCount('students'));
    }

    /** DELETE /api/admin/programs/{program} */
    public function destroy(Request $request, Program $program): Response|JsonResponse
    {
        $blocked = $this->blockedMessage($program->name, [
            'student' => $program->students()->count(),
            'course' => $program->courses()->count(),
        ]);
        if ($blocked) {
            return ApiResponse::message($blocked, 422);
        }

        if ($failure = $this->deleteOrBlock($program, $program->name)) {
            return $failure;
        }

        $this->audit->record($request->user(), 'RECORD_DELETED', "Program: {$program->name}",
            "Deleted program {$program->name} ({$program->award})", null, ['id' => $program->id]);

        return response()->noContent();
    }
}

<?php

namespace App\Http\Resources;

use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\JsonResource;

/**
 * Matches the frontend `User` type. Note the snake_case department_id and
 * department_name: that is what portal.ts expects.
 *
 * @mixin \App\Models\User
 */
class UserResource extends JsonResource
{
    /** @return array<string, mixed> */
    public function toArray(Request $request): array
    {
        $departmentId = $this->resolveDepartmentId();
        $department = match ($this->role->value) {
            'student' => $this->student?->department,
            'lecturer' => $this->lecturer?->department,
            'dept_admin' => $this->department,
            default => null,
        };

        return array_filter([
            'id' => (string) $this->id,
            'name' => $this->name,
            'email' => $this->email,
            'role' => $this->role->value,
            'department_id' => $departmentId !== null ? (string) $departmentId : null,
            'department_name' => $department?->name,
            'status' => $this->status->value,
            'studentId' => $this->student ? (string) $this->student->id : null,
            'lastActive' => $this->last_active_at?->toDateString(),
        ], fn ($value, $key) => ! in_array($key, ['studentId', 'lastActive'], true) || $value !== null, ARRAY_FILTER_USE_BOTH);
    }
}

namespace App\Models;

use App\Enums\UserRole;
use App\Enums\UserStatus;
use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\HasOne;
use Illuminate\Foundation\Auth\User as Authenticatable;
use Illuminate\Notifications\Notifiable;
use Laravel\Sanctum\HasApiTokens;

class User extends Authenticatable
{
    use HasApiTokens, HasFactory, Notifiable;

    /**
     * Role, status and department are fillable for seeders and services only.
     * Controllers must pass explicit validated fields, never $request->all().
     *
     * @var list<string>
     */
    protected $fillable = [
        'name',
        'email',
        'password',
        'role',
        'status',
        'department_id',
        'last_active_at',
    ];

    /** @var list<string> */
    protected $hidden = [
        'password',
        'remember_token',
    ];

    /** @return array<string, string> */
    protected function casts(): array
    {
        return [
            'email_verified_at' => 'datetime',
            'last_active_at' => 'datetime',
            'password' => 'hashed',
            'role' => UserRole::class,
            'status' => UserStatus::class,
        ];
    }

    /** Set for department admins (their scope). Students and lecturers use their profile's department. */
    public function department(): BelongsTo
    {
        return $this->belongsTo(Department::class);
    }

    public function student(): HasOne
    {
        return $this->hasOne(Student::class);
    }

    public function lecturer(): HasOne
    {
        return $this->hasOne(Lecturer::class);
    }

    public function hasRole(UserRole ...$roles): bool
    {
        return in_array($this->role, $roles, true);
    }

    public function isActive(): bool
    {
        return $this->status === UserStatus::Active;
    }

    /**
     * The department this user belongs to, whatever their role.
     * Department admins use users.department_id; students and lecturers use their profile row.
     */
    public function resolveDepartmentId(): ?int
    {
        return match ($this->role) {
            UserRole::Student => $this->student?->department_id,
            UserRole::Lecturer => $this->lecturer?->department_id,
            UserRole::DeptAdmin => $this->department_id,
            default => null,
        };
    }
}

<?php

namespace App\Support;

use App\Enums\UserRole;
use App\Models\User;
use Illuminate\Database\Eloquent\Builder;

/**
 * Department scoping for department admins. It fails closed: any role that is
 * not super admin or dept admin (or a dept admin without a department) sees nothing.
 */
class DepartmentScope
{
    /** Restrict a query to the departments the user may see. */
    public static function apply(Builder $query, User $user, string $column = 'department_id'): Builder
    {
        if ($user->role === UserRole::SuperAdmin) {
            return $query;
        }

        if ($user->role === UserRole::DeptAdmin && $user->department_id) {
            return $query->where($column, $user->department_id);
        }

        return $query->whereRaw('1 = 0');
    }

    public static function canAccess(User $user, ?int $departmentId): bool
    {
        if ($user->role === UserRole::SuperAdmin) {
            return true;
        }

        return $user->role === UserRole::DeptAdmin
            && $user->department_id !== null
            && $departmentId !== null
            && (int) $user->department_id === (int) $departmentId;
    }

    /** Use on every single-record action. Aborts with 403 when the record is in another department. */
    public static function authorize(User $user, ?int $departmentId): void
    {
        abort_unless(self::canAccess($user, $departmentId), 403, 'This record belongs to another department.');
    }
}

namespace Tests\Feature\Admin;

use App\Enums\UserRole;

use Tests\Concerns\CreatesDemoData;

class UserAdminApiTest extends TestCase
{
    use CreatesDemoData;

    public function test_only_super_admins_can_use_the_user_endpoints(): void
    {
        $department = $this->makeDepartment();
        $target = $this->makeStudent($department);

        $this->getJson('/api/admin/users')->assertUnauthorized();

        foreach ([$this->makeDeptAdmin($department), $this->makeLecturer($department), $this->makeStudent($department)] as $user) {
            $this->getJson('/api/admin/users', $this->bearer($user))->assertForbidden();

        }

        $this->assertSame(UserStatus::Active, $target->fresh()->status);
    }

    public function test_list_returns_users_and_supports_search_role_and_status_filters(): void
    {
        $department = $this->makeDepartment();
        $admin = $this->makeSuperAdmin(['name' => 'Morgan Ellis']);
        $lecturer = $this->makeLecturer($department, ['name' => 'Dr. Maya Chen']);
        $this->makeStudent($department, ['name' => 'Ethan Okafor', 'status' => UserStatus::Suspended]);

        $all = $this->getJson('/api/admin/users', $this->bearer($admin))
            ->assertOk()
            ->assertJsonPath('meta.total', 3)
            ->assertJsonStructure(['data' => [['id', 'name', 'email', 'role', 'department_id', 'department_name', 'status']], 'meta' => ['current_page', 'last_page', 'total']]);

            ->assertOk()
            ->assertJsonPath('meta.total', 1)
            ->assertJsonPath('data.0.email', $lecturer->email)
            ->assertJsonPath('data.0.department_name', 'Computer Science');

        $this->getJson('/api/admin/users?status=suspended', $this->bearer($admin))
            ->assertOk()

let memorySession: AuthSession | null = null;
let memoryOnly = false;

export function isSession(value: unknown): value is AuthSession {
  if (!value || typeof value !== "object") return false;
  const session = value as AuthSession;
  const user = session.user;
  return typeof session.token === "string" && !!session.token.trim() && !!user && typeof user === "object" &&
    typeof user.id === "string" && !!user.id.trim() && typeof user.name === "string" &&
    typeof user.email === "string" && ROLES.includes(user.role) &&
    (user.department_id === null || typeof user.department_id === "string") &&
    (user.department_name === null || typeof user.department_name === "string") &&
    STATUSES.includes(user.status) &&
    (user.studentId === undefined || typeof user.studentId === "string") &&
    (user.avatarUrl === undefined || typeof user.avatarUrl === "string");
}

/** Reads the current session, including sessions changed by another tab. */
export function getSession(): AuthSession | null {
  if (typeof window === "undefined" || memoryOnly) return memorySession;
  let raw: string | null;
  try {
    raw = window.localStorage.getItem(AUTH_STORAGE_KEY);
  } catch {
    // Storage may be disabled; the in-memory session still works for this page.
    return memorySession;
  }
  if (!raw) return (memorySession = null);
  try {
    const parsed: unknown = JSON.parse(raw);
    return (memorySession = isSession(parsed) ? parsed : null);
  } catch {
    return (memorySession = null);
  }
}

export function setSession(session: AuthSession | null): void {
  if (session !== null && !isSession(session)) throw new Error("Invalid authentication session");
  memorySession = session;
  if (typeof window === "undefined") return;
  try {
    if (session) window.localStorage.setItem(AUTH_STORAGE_KEY, JSON.stringify(session));
    else window.localStorage.removeItem(AUTH_STORAGE_KEY);
    memoryOnly = false;
  } catch {
    // Keep the in-memory session when storage is unavailable.
    memoryOnly = true;
  }
  window.dispatchEvent(new Event(SESSION_EVENT));
}

export function getToken(): string | null {
  return getSession()?.token ?? null;
}

/** Subscribe to both same-tab and cross-tab session changes. */
export function subscribeToSession(listener: () => void): () => void {
  if (typeof window === "undefined") return () => {};
  const onStorage = (event: StorageEvent) => {
    if (event.key === AUTH_STORAGE_KEY || event.key === null) {
      memoryOnly = false;
      listener();
    }
  };
  window.addEventListener(SESSION_EVENT, listener);
  window.addEventListener("storage", onStorage);
  return () => {
    window.removeEventListener(SESSION_EVENT, listener);
    window.removeEventListener("storage", onStorage);
  };
}

/** Field errors from a Laravel 422 response: { errors: { field: ["message"] } }. */
export type FieldErrors = Record<string, string[]>;

export class ApiError extends Error {
  constructor(public status: number, message: string, public details?: unknown) {
    super(message);
    this.name = "ApiError";
  }
  get isUnauthorized(): boolean { return this.status === 401; }
  get isForbidden(): boolean { return this.status === 403; }
  get isValidation(): boolean { return this.status === 422; }
  /** Per-field messages for a 422 response, or an empty object. */
  get fieldErrors(): FieldErrors {
    const errors = (this.details as { errors?: unknown } | undefined)?.errors;
    return errors && typeof errors === "object" ? (errors as FieldErrors) : {};
  }
}

export interface ApiRequestOptions extends Omit<RequestInit, "body"> {
  /** JSON-serializable request body. */
  body?: unknown;
  /** Override the stored token; pass null for an unauthenticated request. */
  token?: string | null;
}

/** Generic JSON request. A 401 clears the matching session; a 403 keeps it. */
export async function apiRequest<T>(path: string, options: ApiRequestOptions = {}): Promise<T> {
  const { body, token: tokenOverride, ...init } = options;
  const token = tokenOverride === undefined ? getToken() : tokenOverride;
  const normalizedPath = path.startsWith("/") ? path : `/${path}`;
  const clearExpiredSession = (status: number) => {
    // Do not erase a newer login if an older request completes late.
    if (status === 401 && token && getToken() === token) setSession(null);
  };

  if (USE_MOCK) {
    try {
      return mockRequest(normalizedPath, { method: init.method, token, body }) as T;
    } catch (error) {
      if (error instanceof MockApiError) {
        clearExpiredSession(error.status);
        throw new ApiError(error.status, error.message);
      }
      throw error;
    }
  }
  if (!API_URL) throw new ApiError(0, "API URL is not configured and mock mode is disabled");
  const headers = new Headers(init.headers);
  headers.set("Accept", "application/json");
  if (body !== undefined && !headers.has("Content-Type")) headers.set("Content-Type", "application/json");
  if (token) headers.set("Authorization", `Bearer ${token}`);
  let response: Response;
  try {
    response = await fetch(`${API_URL}${normalizedPath}`, {
      ...init,
      headers,
      body: body === undefined ? undefined : JSON.stringify(body),
    });
  } catch {
    throw new ApiError(0, "Cannot reach the server. Check your connection and try again.");
  }
  const text = await response.text();
  let data: unknown;
  if (text) {
    try { data = JSON.parse(text); } catch { data = text; }
  }
  if (!response.ok) {
    clearExpiredSession(response.status);
    const message = typeof data === "object" && data !== null && "message" in data && typeof data.message === "string"
      ? data.message : typeof data === "string" && data ? data : response.statusText || `Request failed (${response.status})`;
    throw new ApiError(response.status, message, data);
  }
  return data as T;
}

/** Builds a query string, skipping empty values. */
export function toQuery(params: Record<string, string | number | undefined | null> = {}): string {
  const search = new URLSearchParams();
  Object.entries(params).forEach(([key, value]) => {
    if (value !== undefined && value !== null && value !== "") search.set(key, String(value));
  });
  const query = search.toString();
  return query ? `?${query}` : "";
}

<?php

namespace App\Http\Requests\Admin;

use App\Enums\UserRole;
use Illuminate\Foundation\Http\FormRequest;

/**
 * Base for super admin write requests.
 * - The route group already requires role:super_admin; authorize() repeats the check (defence in depth).
 * - Input keys are camelCase, so validation errors come back under camelCase keys,
 *   which is what the frontend forms read.
 */
abstract class AdminRequest extends FormRequest
{
    /** Input keys whose text is trimmed before validation. @var list<string> */
    protected array $trim = [];

    public function authorize(): bool
    {
        return $this->user()?->role === UserRole::SuperAdmin;
    }

    protected function prepareForValidation(): void
    {
        $clean = [];

        foreach ($this->trim as $key) {
            $value = $this->input($key);

            if (is_string($value)) {
                $value = trim($value);
                // Optional text fields arrive as null or "". Both mean "not set".
                $clean[$key] = $value === '' ? null : $value;
            }
        }

        $this->merge($clean);
    }

    /** Id sent by the frontend as a string, as an integer for queries (null when not numeric). */
    protected function idInput(string $key): ?int
    {
        $value = $this->input($key);

        return is_int($value) || (is_string($value) && ctype_digit($value)) ? (int) $value : null;
    }
}

            $duplicate = Semester::where('academic_session_id', $session->id)
                ->where('name', $this->input('name'))
                ->when($current, fn ($q) => $q->where('id', '!=', $current->id))
                ->exists();

            if ($duplicate) {
                $validator->errors()->add('name', 'This session already has a '.$this->input('name').' semester.');

{

    {
    }

    /** GET /api/dept/submissions?status=&search=&page=&per_page= */
    public function index(Request $request): array
    {
        // select() must come BEFORE withCount(): a later select() would drop the count column.
        $query = ResultSubmission::query()
            ->select('result_submissions.*')
            ->with($this->relations())
            ->withCount('results as student_count')
            ->join('courses', 'courses.id', '=', 'result_submissions.course_id')
            ->orderByDesc('result_submissions.updated_at')
            ->orderByDesc('result_submissions.id');

        DepartmentScope::apply($query, $request->user(), 'courses.department_id');
        $this->search($query, $request, ['courses.code', 'courses.title']);

        $status = $request->query('status');
        if (is_string($status) && ResultStatus::tryFrom($status)) {
            $query->where('result_submissions.status', $status);
        }

        return ApiResponse::paginated($query->paginate(ApiResponse::perPage($request)), SubmissionResource::class);
    }

    /** GET /api/dept/submissions/{submission} */
    public function show(Request $request, ResultSubmission $submission): SubmissionResource
    {
        $this->authorizeSubmission($request, $submission);

        return new SubmissionResource($this->loaded($submission->getKey()));
    }

    /** GET /api/dept/submissions/{submission}/results: the scores the admin is reviewing. */
    public function results(Request $request, ResultSubmission $submission): array
    {
        $this->authorizeSubmission($request, $submission);

        $rows = $submission->results()->with('student.user')->get()
            ->sortBy(fn ($result) => $result->student?->matric_no ?? '')
            ->values()
            ->map(fn ($result) => [
                'id' => (string) $result->id,
                'studentId' => (string) $result->student_id,
                'matricNo' => $result->student?->matric_no ?? '',
                'studentName' => $result->student?->user?->name ?? '',
                'caScore' => $result->ca_score !== null ? (float) $result->ca_score : null,
                'examScore' => $result->exam_score !== null ? (float) $result->exam_score : null,
                'totalScore' => $result->total_score !== null ? (float) $result->total_score : null,
                'grade' => $result->grade,
                'gradePoint' => $result->grade_point !== null ? (float) $result->grade_point : null,
            ])
            ->all();

        return ['data' => $rows];
    }

    /** POST /api/dept/submissions/{submission}/approve */
    public function approve(Request $request, ResultSubmission $submission): SubmissionResource
    {
        return $this->review(
            $request, $submission,
            [ResultStatus::Submitted], ResultStatus::Approved,
            'RESULT_APPROVED', 'Approved results for %s',
            'Only submitted results can be approved.', null,
        );
    }

    /** POST /api/dept/submissions/{submission}/reject  body: { feedback } (required) */
    public function reject(Request $request, ResultSubmission $submission): SubmissionResource
    {
        $this->authorizeSubmission($request, $submission);
        $data = $request->validate(['feedback' => ['required', 'string', 'min:3', 'max:1000']]);

        return $this->review(
            $request, $submission,
            [ResultStatus::Submitted], ResultStatus::Rejected,
            'RESULT_REJECTED', 'Rejected results for %s and returned them to the lecturer',
            'Only submitted results can be rejected.', $data['feedback'],
        );
    }

    /** POST /api/dept/submissions/{submission}/reopen  body: { feedback } (optional reason) */
    public function reopen(Request $request, ResultSubmission $submission): SubmissionResource
    {
        $this->authorizeSubmission($request, $submission);
        $data = $request->validate(['feedback' => ['nullable', 'string', 'max:1000']]);

        return $this->review(
            $request, $submission,
            [ResultStatus::Approved], ResultStatus::Reopened,
            'RESULT_REOPENED', 'Reopened approved results for %s so the lecturer can edit them',
            'Only approved results can be reopened.', $data['feedback'] ?? null,
        );
    }

    /**
     * Move a submission to a new status inside one transaction. The row is locked and the
     * status is checked again, so two admins cannot approve and reject the same submission.
     *
     * @param  list<ResultStatus>  $allowedFrom
     */
    private function review(
        Request $request,
        ResultSubmission $submission,
        array $allowedFrom,
        ResultStatus $to,
        string $action,
        string $detailsTemplate,
        string $failMessage,
        ?string $note,
    ): SubmissionResource {
        $user = $request->user();
        $this->authorizeSubmission($request, $submission);
        $code = $submission->course->code;

        DB::transaction(function () use ($user, $submission, $allowedFrom, $to, $action, $detailsTemplate, $failMessage, $note, $code) {
            $locked = ResultSubmission::query()->whereKey($submission->getKey())->lockForUpdate()->firstOrFail();

            if (! in_array($locked->status, $allowedFrom, true)) {
                abort(422, $failMessage);
            }

            $locked->update([
                'status' => $to,
                'feedback' => $note,
                'reviewed_by' => $user->id,
                'reviewed_at' => now(),
            ]);

            $this->audit->record(
                $user,
                $action,
                $code,
                sprintf($detailsTemplate, $code),
                $locked,
                $note !== null ? ['note' => $note] : [],
            );
        });

        return new SubmissionResource($this->loaded($submission->getKey()));
    }

    /** 403 unless the submission's course belongs to the admin's own department. */
    private function authorizeSubmission(Request $request, ResultSubmission $submission): void
    {
        $submission->loadMissing('course');
        DepartmentScope::authorize($request->user(), $submission->course?->department_id);
    }

    private function loaded(int|string $id): ResultSubmission
    {
        return ResultSubmission::query()
            ->with($this->relations())
            ->withCount('results as student_count')
            ->findOrFail($id);
    }

    /** @return array<int|string, mixed> */
    private function relations(): array
    {
        return [
            'course.department',
            'lecturer.user',
            'semester.session',
            'auditLogs' => fn ($query) => $query->orderBy('created_at')->orderBy('id'),
        ];
    }
}

/**
 * Shared helpers for the department admin controllers (T2.6).
 * Every query in these controllers must go through DepartmentScope::apply() first.
 */
abstract class DeptAdminController extends Controller
{
    /** Apply an exact-match filter when the query value is a whole number (ids and levels). */
    protected function whereId(Builder $query, Request $request, string $param, string $column): void
    {
        $value = $request->query($param);
        if (is_string($value) && ctype_digit($value)) {
            $query->where($column, (int) $value);
        }
    }

    /**
     * Case-insensitive search. It is always wrapped in its own group so an "or" inside the
     * search can never switch off the department scope that was added before it.
     *
     * @param  list<string>  $columns
     */
    protected function search(Builder $query, Request $request, array $columns): void
    {
        $term = ApiResponse::search($request);
        if ($term === null || $term === '') {
            return;
        }

        $query->where(fn (Builder $group) => LikeSearch::apply($group, $term, $columns));
    }
